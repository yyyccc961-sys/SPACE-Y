-- صلاحيات الكتابة للوحة الإدارة مرتبطة بإيميل المدير المحدد أدناه.

CREATE TABLE IF NOT EXISTS public.sales (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  customer_name TEXT,
  item_name TEXT NOT NULL,
  amount NUMERIC(12,2) NOT NULL CHECK (amount > 0),
  amount_received NUMERIC(12,2) NOT NULL DEFAULT 0 CHECK (amount_received >= 0 AND amount_received <= amount),
  payment_status TEXT NOT NULL DEFAULT 'pending' CHECK (payment_status IN ('paid','pending','partial')),
  sale_date DATE NOT NULL DEFAULT CURRENT_DATE,
  notes TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
ALTER TABLE public.sales ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "SPACE admin read sales" ON public.sales;
DROP POLICY IF EXISTS "SPACE admin add sales" ON public.sales;
DROP POLICY IF EXISTS "SPACE admin delete sales" ON public.sales;
CREATE POLICY "SPACE admin read sales" ON public.sales FOR SELECT TO authenticated USING ((auth.jwt()->>'email') = 'space.admin@outlook.com');
CREATE POLICY "SPACE admin add sales" ON public.sales FOR INSERT TO authenticated WITH CHECK ((auth.jwt()->>'email') = 'space.admin@outlook.com');
CREATE POLICY "SPACE admin delete sales" ON public.sales FOR DELETE TO authenticated USING ((auth.jwt()->>'email') = 'space.admin@outlook.com');
